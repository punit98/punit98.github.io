hourly = []
day_single = []
week_single = []
month_single = []

day_moving_avg = []
week_moving_avg = []
month_moving_avg = []

random.seed(0)
for i in range(12): # 12 months per year
    for j in range(4): # 4 weeks per month
        for k in range(7): # 7 days per week
            for l in range(24): # 24 hours per day
                percent_ava = random.randint(95,101)
                hourly.append(percent_ava)   
            day_moving_avg.append(math.ceil(sum(hourly) / len(hourly))) # moving average
            day_single.append(percent_ava) # last value received per day
        week_moving_avg.append(math.ceil(sum(day_moving_avg) / len(day_moving_avg))) # moving average
        week_single.append(percent_ava) # last value received per week
    month_moving_avg.append(math.ceil(sum(week_moving_avg) / len(week_moving_avg))) # moving average
    month_single.append(percent_ava) # last value received per month

def chunk(lst, num):
    for i in range(0, len(lst), num): 
        yield lst[i:i + num]

day_list = list(chunk(hours_record, 24))
daily_record = []
for i in day_list:
    counter = Counter(i)
    daily_record.append(counter.most_common(1)[0][0])
    
    
week_list = list(chunk(daily_record, 7))
weekly_record = []
for i in week_list:
    counter = Counter(i)
    weekly_record.append(counter.most_common(1)[0][0])
    
    
month_list = list(chunk(weekly_record, 4))
monthly_record = []
for i in month_list:
    counter = Counter(i)
    monthly_record.append(counter.most_common(1)[0][0])
    

daily_month = list(chunk(daily_record, 28))
month_daily_record = []
for i in daily_month:
    counter = Counter(i)
    month_daily_record.append(counter.most_common(1)[0][0])

    
day_avg = []
for i in range(len(daily_record)):
    day_avg.append(median(daily_record))
    
week_avg = []
for i in range(len(weekly_record)):
    week_avg.append(median(weekly_record))
    
month_avg = []
for i in range(len(monthly_record)):
    month_avg.append(median(monthly_record))
    
daily_month_avg = []
for i in range(len(month_daily_record)):
    daily_month_avg.append(median(month_daily_record))

I'd suggest using median whenever possible because mean is affected by outlier values. see graph below.¶
